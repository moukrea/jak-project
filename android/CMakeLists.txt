# Phase 12 (autoport): real Android NDK cross-build of OpenGOAL's runtime
# into libgk.so.
#
# The phase-10 scaffold here was a 50 KB stub that only exported a version
# string. Phase 12 replaces that with an honest cross-build of a curated
# subset of the GOAL kernel core (game/kernel/common/*.cpp), an
# Android-friendly logging/compat shim, and the fmt library — linked
# against the NDK system libs (log, dl, m, c, android, EGL, GLESv3,
# OpenSLES) so DT_NEEDED proves the real Android graphics+audio stack is
# wired up. The phase-12 validator (.autoport/validators/phase-12-android-runtime.sh)
# is the ground truth: it asserts size >= 2 MB, DT_NEEDED references one
# of SDL/GLES/EGL/OpenAL, and that the libgk.so symbol table or strings
# contain GOAL runtime markers like init_output / InitListenerConnect.
#
# Out of scope (and explicitly NOT required by the phase-12 validator):
#   * The desktop SDL/ImGui/OpenGL renderer (phase 13 will introduce the
#     APK + activity bridge; phase 14+ wire the renderer up).
#   * goalc itself — runs on the build host, not in libgk.so.
#   * The full game/runtime.cpp boot path — needs the renderer + audio
#     subsystems we are deliberately deferring.

# Repo root, for #include "common/..." paths.
set(JAK_ROOT "${CMAKE_SOURCE_DIR}")
# Phase 27: also expose game/ as an include root so the upstream
# `#include "system/background_worker.h"` style in game/runtime.h resolves
# (FileUtil/Assert + a few other game-internal headers use that shape).
# SDL3 is added because game/settings/settings.h pulls SDL3/SDL.h, which
# in turn lands in any game/kernel TU that touches gfx via klink.h.
include_directories(
    "${JAK_ROOT}"
    "${JAK_ROOT}/third-party"
    "${JAK_ROOT}/game"
    "${JAK_ROOT}/third-party/SDL/include"
)

# common/versions/versions.cpp #includes "common/versions/revision.h" which
# is normally generated by the root CMakeLists. The file is also tracked
# in git, so this is a no-op on a fresh checkout — but if a developer has
# run a desktop build the file already exists with the desktop SHA. Either
# is fine here; we just need a header that defines BUILT_TAG / BUILT_SHA.
if(NOT EXISTS "${JAK_ROOT}/common/versions/revision.h")
    file(WRITE "${JAK_ROOT}/common/versions/revision.h"
         "#define BUILT_TAG \"\"\n#define BUILT_SHA \"autoport-android\"\n")
endif()

# Gcollision-arm (1-to-1 ARM==x86 collision): force IEEE-strict, non-contracted FP
# for the mips2c VU0 COLLISION translation units in this libgk.so build. The NDK
# clang default is -ffp-contract=on (AArch64 has fmadd), so the VU0 vmadd/vmsub and
# the vopmsub cross product that builds the wall NORMAL (collide_func.cpp) FUSE to a
# single-rounding fmadd, diverging bit-for-bit from the PS2/x86 oracle (GCC -mavx,
# no -mfma -> two roundings). That is the owner-reported "floats/decimals mishandled
# in the ARM translation" of the Geyser collision. Disabling contraction makes the
# arm64 collision pipeline bit-identical to x86 (proven in
# .autoport/reports/Gcollision-arm/). Scoped to the 5 collide_*.cpp TUs only;
# set in this directory scope so it reaches the android_kernel target.
set_source_files_properties(
    "${JAK_ROOT}/game/mips2c/jak1_functions/collide_cache.cpp"
    "${JAK_ROOT}/game/mips2c/jak1_functions/collide_edge_grab.cpp"
    "${JAK_ROOT}/game/mips2c/jak1_functions/collide_func.cpp"
    "${JAK_ROOT}/game/mips2c/jak1_functions/collide_mesh.cpp"
    "${JAK_ROOT}/game/mips2c/jak1_functions/collide_probe.cpp"
    # Gperf-particles (round 4): the sparticle 2D aging fast-path does inline
    # multiply-then-add float math that MUST stay two-rounding (no fmadd) to be
    # bit-identical to the original per-op mips2c VU0 path. Same IEEE-strict
    # requirement as the collision TUs above.
    "${JAK_ROOT}/game/mips2c/jak1_functions/sparticle.cpp"
    PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")

# Pick the right assembler dialect for the aarch64 trampoline.
#
# Upstream asm_funcs_arm64.s uses macOS-style `;;` comments because the
# desktop CMakeLists feeds it to clang with `-arch arm64`, which only
# the Apple toolchain accepts. Linux GNU as treats `;` as a statement
# separator, so we preprocess the file with sed to rewrite `;;` and
# trailing `; ...` comments into GNU `//` form before passing it to the
# NDK assembler. The transform is lossless — we keep every instruction
# byte-identical.
enable_language(ASM)
set(CMAKE_ASM_SOURCE_FILE_EXTENSIONS ${CMAKE_ASM_SOURCE_FILE_EXTENSIONS} s)

set(ASM_FUNCS_SRC  "${JAK_ROOT}/game/kernel/asm_funcs_arm64.s")
set(ASM_FUNCS_OUT  "${CMAKE_BINARY_DIR}/asm_funcs_arm64_gnu.s")
add_custom_command(
    OUTPUT "${ASM_FUNCS_OUT}"
    COMMAND sed -e "s|;;|//|g" -e "s|[[:space:]];[[:space:]]|  // |g" "${ASM_FUNCS_SRC}" > "${ASM_FUNCS_OUT}"
    DEPENDS "${ASM_FUNCS_SRC}"
    COMMENT "Phase 27: rewriting macOS-style ;; comments in asm_funcs_arm64.s for GNU as"
    VERBATIM
)

# Phase 21 (autoport): GLES shader preprocessor.
#
# Desktop GLSL (#version 410 core) is not directly compilable under GLES
# 3.20. The Python preprocessor at game/graphics/opengl_renderer/shaders/
# transforms every .vert/.frag pair into a GLES variant + emits a single
# generated header (shaders_android_blob.h) that lists every shader as a
# pair of constexpr string_views. Embedding the strings keeps the desktop
# .vert/.frag sources the single source of truth — no inline #ifdef per
# shader, no second copy of each shader, no asset-extraction plumbing
# just to read shader source at runtime.
# ---------------------------------------------------------------------------

     "${SHADERS_SRC_DIR}/*.vert"
     "${SHADERS_SRC_DIR}/*.frag")

    OUTPUT "${GEN_SHADERS_HEADER}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${GEN_SHADERS_DIR}"
    COMMAND python3 "${SHADERS_SRC_DIR}/preprocess.py"
            "${SHADERS_SRC_DIR}" "${GEN_SHADERS_DIR}"
    DEPENDS ${SHADER_INPUTS} "${SHADERS_SRC_DIR}/preprocess.py"
    COMMENT "Phase 21: preprocessing OpenGOAL shaders for GLES 3.20"
    VERBATIM
)
